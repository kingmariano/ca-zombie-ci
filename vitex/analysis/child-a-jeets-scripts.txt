    async function callWithTimeout(url, method, params, ms=12000){
      const controller = new AbortController();
      const t = setTimeout(()=>controller.abort(), ms);
      try{
        const r = await fetch(url, {
          method: "POST",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({ jsonrpc:"2.0", id:1, method, params }),
          mode: "cors",
          signal: controller.signal
        });
        if(!r.ok) throw new Error(`HTTP ${r.status}`);
        const j = await r.json();
        if(j?.error) throw new Error(`RPC ${j.error.code}: ${j.error.message}`);
        return j.result;
      } finally { clearTimeout(t); }
    }

    async function rpcCall(method, params){
      const errs = [];
      for(const url of RPCS){
        try{
          return await callWithTimeout(url, method, params, 12000);
        }catch(e){
          errs.push(`${url} → ${e.message}`);
        }
      }
      const err = new Error("All RPCs failed");
      err.details = errs;
      throw err;
    }

    async function getJeetsBalance(owner){
      if(!window.JEETS_MINT) throw new Error("JEETS mint not set");
      const res = await rpcCall("getTokenAccountsByOwner", [
        owner,
        { mint: window.JEETS_MINT },
        { encoding: "jsonParsed", commitment: "confirmed" }
      ]);
      const accounts = res?.value || [];
      let total = 0, decimals = 0;
      for(const a of accounts){
        const info = a?.account?.data?.parsed?.info?.tokenAmount;
        if(info){ decimals = info.decimals; total += Number(info.uiAmount || 0); }
      }
      return { total, decimals };
    }
